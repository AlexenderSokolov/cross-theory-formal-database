\begin{algorithm}[t]
    \caption{$\SATALG$}\label[algorithm]{alg:start}
    \begin{small}
        \Function{$\SATALG(\varphi : \text{node expression})$}
        {
            {\bf require:} true \\
            $\NOM \gets \text{a new nominal}$\;
            $\Phi\gets [\NOM{:}\varphi]$\;
            \Return{$\SATALG(\Phi,\NOM+1)$}\;
            {\bf ensure:} $\SATALG(\varphi)$ iff $\varphi$ is satisfiable \\
        }
    \end{small}
\end{algorithm}
