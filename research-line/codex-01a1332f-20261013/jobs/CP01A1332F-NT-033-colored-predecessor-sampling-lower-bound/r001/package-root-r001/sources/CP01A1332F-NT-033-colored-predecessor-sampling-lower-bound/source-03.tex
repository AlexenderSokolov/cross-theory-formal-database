\paragraph{Theorem 8.}
There exists a constant $c$ such that for $r=cq$ the following holds.
\begin{itemize}
\item Let $f:W^N\to(\{\widehat O,\widehat *,O,*\}^r)^m$ be a depth-$q$ forest with word size $w\geq\log m$.
\item Let $\Pi$ be an $r\times m$ random matrix as in Definition 6.
\item Then $\Delta(f(U_{W^N}),\operatorname{CMPred}(\Pi))\geq1-2\cdot2^{-m/w^{O(q)}}$.
\end{itemize}
