We continue with the assumptions of Sections 2--3; but from now on (except in Remark 4.17) for simplicity we assume that $\Bbbk$ is a field.

\subsection*{4.3. Tilting objects in the Satake category}
Recall the notion of parity complexes in $D^{\mathrm b}_{\jepPubEuler{IW}}(\mathrm{Gr},\Bbbk)$ considered in Remark~\ref{jep104local027}.
In certain proofs of this subsection we will also consider the $(I^+_{\mathrm u},\chi_{I^+}^*(\jepPubEuler{L}^{\Bbbk}_{\psi}))$-equivariant constructible derived category of $\mathrm{Fl}$, which we will denote $D^{\mathrm b}_{\jepPubEuler{IW}}(\mathrm{Fl},\Bbbk)$.
Of course, we can also consider parity complexes in this category, as well as in the $I^-$-equivariant derived category $D^{\mathrm b}_{I^-}(\mathrm{Fl},\Bbbk)$, or in the $G_{\jepPubScript{O}}$-constructible derived category $D^{\mathrm b}_{(G_{\jepPubScript{O}})}(\mathrm{Gr},\Bbbk)$.
(Note that, by definition, an object of $D^{\mathrm b}_{I^-}(\mathrm{Fl},\Bbbk)$ is parity iff its image in the $I^-$-constructible derived category $D^{\mathrm b}_{(I^-)}(\mathrm{Fl},\Bbbk)$ is parity.)
In particular, for any $\lambda\in\boldsymbol{X}^{\vee}_+$, we denote by $\mathcal E_{\lambda}$ the unique indecomposable parity complex in the category $D^{\mathrm b}_{(G_{\jepPubScript{O}})}(\mathrm{Gr},\Bbbk)$ supported on $\overline{\mathrm{Gr}^{\lambda}}$ and whose restriction to $\mathrm{Gr}^{\lambda}$ is $\underline{\Bbbk}_{\mathrm{Gr}^{\lambda}}[\dim\mathrm{Gr}^{\lambda}]$ (see~\cite[Th.~2.12 \& \S4.1]{jep104ref31}).
Since $\mathsf H^{\bullet}_{G_{\jepPubScript{O}}}(\mathrm{pt};\Bbbk)$ might not be concentrated in even degrees, in general the theory of~\cite{jep104ref31} does not apply in $D^{\mathrm b}_{G_{\jepPubScript{O}}}(\mathrm{Gr},\Bbbk)$. This difficulty will be remedied by the following lemma.
\begin{lem}\label{jep104new048}
Any parity object $\mathcal E$ in $D^{\mathrm b}_{(G_{\jepPubScript{O}})}(\mathrm{Gr},\Bbbk)$ is a direct summand of a parity object $\mathcal E'$ which belongs to the essential image of the functor $\mathsf{For}_{G_{\jepPubScript{O}}}:D^{\mathrm b}_{G_{\jepPubScript{O}}}(\mathrm{Gr},\Bbbk)\to D^{\mathrm b}_{(G_{\jepPubScript{O}})}(\mathrm{Gr},\Bbbk)$.
\end{lem}
\begin{proof}
Of course we can assume that $\mathcal E=\mathcal E_{\lambda}$ for some $\lambda\in\boldsymbol{X}^{\vee}_+$.
Recall that the forgetful functor $\mathsf{For}_{I^-}$ sends indecomposable parity objects to indecomposable parity objects (see~\cite[Lem.~2.4]{jep104ref35}).
In view of the classification of such objects in the $I^-$-equivariant and $I^-$-constructible derived categories, this means that any $I^-$-constructible parity complex on $\mathrm{Gr}$ belongs to the essential image of $\mathsf{For}_{I^-}$.
In particular, there exists a parity complex $\jepPubEuler{F}$ in $D^{\mathrm b}_{I^-}(\mathrm{Gr},\Bbbk)$ such that $\mathcal E_{\lambda}\cong\mathsf{For}_{I^-}(\jepPubEuler{F})$.
Now we set $\mathcal E':=\mathsf{For}_{G_{\jepPubScript{O}}}({}^{*}\mathsf{Ind}_{I^-}^{G_{\jepPubScript{O}}}(\jepPubEuler{F}))$.
Then $\mathcal E'$ is parity as a convolution of parity complexes, see~\cite[Th.~4.8]{jep104ref31}.
And since this object is supported on $\overline{\mathrm{Gr}^{\lambda}}$ and has nonzero restriction to $\mathrm{Gr}^{\lambda}$, it must admit a cohomological shift of $\mathcal E_{\lambda}$ as a direct summand.
\end{proof}
Recall that the forgetful functor $\mathsf{For}_{G_{\jepPubScript{O}}}:D^{\mathrm b}_{G_{\jepPubScript{O}}}(\mathrm{Gr},\Bbbk)\to D^{\mathrm b}_{(G_{\jepPubScript{O}})}(\mathrm{Gr},\Bbbk)$ restricts to an equivalence between $G_{\jepPubScript{O}}$-equivariant and $G_{\jepPubScript{O}}$-constructible perverse sheaves, see~\cite[Prop.~2.1]{jep104ref36} (or~\cite[Prop.~1.10.8]{jep104ref11}).
Therefore for any $\jepPubEuler{F}$ in $D^{\mathrm b}_{(G_{\jepPubScript{O}})}(\mathrm{Gr},\Bbbk)$ and any $n\in\mathbb Z$, the perverse sheaf ${}^{p}\jepPubEuler{H}^{n}(\jepPubEuler{F})$ is $G_{\jepPubScript{O}}$-equivariant.
The main result of this section is the following.
\setcounter{thm}{9}
\begin{thm}\label{jep104new0410}
For any $n\in\mathbb Z$ and $\lambda\in\boldsymbol{X}^{\vee}_+$, the $G_{\jepPubScript{O}}$-equivariant perverse sheaf ${}^{p}\jepPubEuler{H}^{n}(\mathcal E_{\lambda})$ is tilting in the highest weight category $\mathrm{Perv}_{G_{\jepPubScript{O}}}(\mathrm{Gr},\Bbbk)$.
In particular, the indecomposable tilting object associated with $\lambda$ is a direct summand of ${}^{p}\jepPubEuler{H}^{0}(\mathcal E_{\lambda})$.
\end{thm}
The proof of Theorem~\ref{jep104new0410} requires a few preliminaries. We start with the following observation, which will be crucial for us.
\setcounter{thm}{11}
\begin{prop}\label{jep104new0412}
The parity objects in $D^{\mathrm b}_{\jepPubEuler{IW}}(\mathrm{Gr},\Bbbk)$ are exactly the direct sums of cohomological shifts of tilting perverse sheaves.
\end{prop}
\begin{proof}
As already noticed in the proof of Corollary~\ref{jep104local001}, the strata $X_{\lambda}\subset\mathrm{Gr}$ supporting $(I^+_{\mathrm u},\chi_{I^+}^*(\jepPubEuler{L}^{\Bbbk}_{\psi}))$-equivariant local systems (i.e., those with $\lambda\in\boldsymbol{X}^{\vee}_{++}$) have dimensions of constant parity on each connected component of $\mathrm{Gr}$.
Therefore, the tilting objects in the highest weight category $\mathrm{Perv}_{\jepPubEuler{IW}}(\mathrm{Gr},\Bbbk)$ are also parity.
By unicity, they must then coincide with the ``parity sheaves'' (or, in another terminology, normalized indecomposable parity complexes) of~\cite[Def.~2.14]{jep104ref31}.
The claim follows, since any parity complex is a direct sum of cohomological shifts of such objects.
\end{proof}
Next we observe that the parity property is preserved under convolution, in the following sense.
\begin{lem}\label{jep104new0413}
If $\jepPubEuler{F}\in D^{\mathrm b}_{\jepPubEuler{IW}}(\mathrm{Fl},\Bbbk)$ and $\jepPubEuler{G}\in D^{\mathrm b}_{I^-}(\mathrm{Fl},\Bbbk)$ are parity complexes, then $\jepPubEuler{F}\star^{I^-}\jepPubEuler{G}\in D^{\mathrm b}_{\jepPubEuler{IW}}(\mathrm{Fl},\Bbbk)$ is a parity complex.
\end{lem}
\begin{proof}
In view of the description of parity complexes in~\cite[\S4.1]{jep104ref31}, the claim follows from standard arguments going back at least to~\cite{jep104ref43}.
In fact it suffices to treat the case $\jepPubEuler{G}=\underline{\Bbbk}_{\overline{\mathrm{Fl}_{w}}}$ when $\ell(w)\in\{0,1\}$, which can be done ``by hand'' as in~\cite{jep104ref43}.
\end{proof}
\begin{lem}\label{jep104new0414}
If $\jepPubEuler{F}\in D^{\mathrm b}_{\jepPubEuler{IW}}(\mathrm{Gr},\Bbbk)$ is parity and $\jepPubEuler{G}\in D^{\mathrm b}_{G_{\jepPubScript{O}}}(\mathrm{Gr},\Bbbk)$ is such that $\mathsf{For}_{G_{\jepPubScript{O}}}(\jepPubEuler{G})$ is parity, then $\jepPubEuler{F}\star^{G_{\jepPubScript{O}}}\jepPubEuler{G}\in D^{\mathrm b}_{\jepPubEuler{IW}}(\mathrm{Gr},\Bbbk)$ is parity.
\end{lem}
\begin{proof}
The natural projection $\pi:\mathrm{Fl}\to\mathrm{Gr}$ (a smooth and projective morphism) is $I^+_{\mathrm u}$-equivariant, hence induces functors
\[
\pi^*:D^{\mathrm b}_{\jepPubEuler{IW}}(\mathrm{Gr},\Bbbk)\to D^{\mathrm b}_{\jepPubEuler{IW}}(\mathrm{Fl},\Bbbk),\qquad
\pi_*:D^{\mathrm b}_{\jepPubEuler{IW}}(\mathrm{Fl},\Bbbk)\to D^{\mathrm b}_{\jepPubEuler{IW}}(\mathrm{Gr},\Bbbk).
\]
The projection formula shows that $\jepPubEuler{F}$ is a direct summand in $\pi_*(\pi^*\jepPubEuler{F})$, and by Lemma~\ref{jep104local017} we have
\[
\pi_*(\pi^*\jepPubEuler{F})\star^{G_{\jepPubScript{O}}}\jepPubEuler{G}\cong\pi^*(\jepPubEuler{F})\star^{I^-}\mathsf{For}^{G_{\jepPubScript{O}}}_{I^-}(\jepPubEuler{G}).
\]
Hence to conclude it suffices to prove that $\pi^*(\jepPubEuler{F})\star^{I^-}\mathsf{For}^{G_{\jepPubScript{O}}}_{I^-}(\jepPubEuler{G})$ is parity. However we have
\[
\pi^*\bigl(\pi^*(\jepPubEuler{F})\star^{I^-}\mathsf{For}^{G_{\jepPubScript{O}}}_{I^-}(\jepPubEuler{G})\bigr)\cong\pi^*(\jepPubEuler{F})\star^{I^-}\pi^*(\mathsf{For}^{G_{\jepPubScript{O}}}_{I^-}(\jepPubEuler{G})).
\]
Since $\pi^*\jepPubEuler{F}$ and $\pi^*(\mathsf{For}^{G_{\jepPubScript{O}}}_{I^-}(\jepPubEuler{G}))$ are parity (because $\pi$ is smooth), Lemma~\ref{jep104new0413} implies that $\pi^*\bigl(\pi^*(\jepPubEuler{F})\star^{I^-}\mathsf{For}^{G_{\jepPubScript{O}}}_{I^-}(\jepPubEuler{G})\bigr)$ is parity.
We deduce that $\pi^*(\jepPubEuler{F})\star^{I^-}\mathsf{For}^{G_{\jepPubScript{O}}}_{I^-}(\jepPubEuler{G})$ is parity, as expected.
\end{proof}
\begin{cor}\label{jep104new0415}
Let $\mathcal E$ be in $D^{\mathrm b}_{G_{\jepPubScript{O}}}(\mathrm{Gr},\Bbbk)$, and assume that $\mathsf{For}_{G_{\jepPubScript{O}}}(\mathcal E)$ is parity.
Then $\Phi(\mathcal E)$ is parity in $D^{\mathrm b}_{\jepPubEuler{IW}}(\mathrm{Gr},\Bbbk)$.
In particular, $\Phi^0({}^{p}\jepPubEuler{H}^{n}(\mathcal E))$ is a tilting perverse sheaf for any $n\in\mathbb Z$.
\end{cor}
\begin{proof}
Since $\Delta^{\jepPubEuler{IW}}_{\varsigma}(\Bbbk)$ is parity (see~\eqref{jep104local003}), the first claim follows from Lemma~\ref{jep104new0414}.
The second claim follows from the facts that $\Phi$ is $t$-exact and that the perverse cohomology objects of parity objects in $D^{\mathrm b}_{\jepPubEuler{IW}}(\mathrm{Gr},\Bbbk)$ are tilting perverse sheaves, see Proposition~\ref{jep104new0412}.
\end{proof}
We can finally give the proof of Theorem~\ref{jep104new0410}.
\begin{proof}[Proof of Theorem~\ref{jep104new0410}]
Since $\Phi^0$ is an equivalence of highest weight categories, to prove the first claim it suffices to prove that $\Phi^0({}^{p}\jepPubEuler{H}^{n}(\mathcal E_{\lambda}))$ is tilting in the highest weight category $\mathrm{Perv}_{\jepPubEuler{IW}}(\mathrm{Gr},\Bbbk)$.
This follows from Lemma~\ref{jep104new048} and Corollary~\ref{jep104new0415}, since a direct summand of a tilting perverse sheaf is tilting.
The second claim follows, since ${}^{p}\jepPubEuler{H}^{0}(\mathcal E_{\lambda})$ is supported on $\overline{\mathrm{Gr}^{\lambda}}$, and has nonzero restriction to $\mathrm{Gr}^{\lambda}$.
\end{proof}
