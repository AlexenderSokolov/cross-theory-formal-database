\section*{Glaeser inequality of order $k$ (the paper's first main question)}

\textbf{Source.} M.~Ghisi, M.~Gobbino, \emph{Higher order Glaeser inequalities and optimal
regularity of roots of real functions}, Ann.\ Sc.\ Norm.\ Super.\ Pisa Cl.\ Sci.\ (5)
\textbf{XII} (2013), 1001--1021; printed page 1002.

\begin{theorem}[as announced in the introduction, inequality (1.2)]
For every integer $k \ge 2$ there exists a constant $C(k)$, depending only upon $k$, such that
\[
|v'(x)|^{k+1} \;\le\; C(k)\cdot |v(x)|^{k}\cdot \sup_{x\in\mathbb{R}} \bigl|v^{(k+1)}(x)\bigr|
\qquad \forall x \in \mathbb{R}
\]
for every $v \in C^{k+1}(\mathbb{R})$ such that either $v(x) \ge 0$ or $v(x) \le 0$ for every
$x \in \mathbb{R}$, and either $v'(x) \ge 0$ or $v'(x) \le 0$ for every $x \in \mathbb{R}$.
\end{theorem}

\begin{remark}[consequence for the $(k+1)$-th root, same page]
Setting $u(x) := |v(x)|^{1/(k+1)}$ (no absolute value needed when $k$ is even), for every
$x \in \mathbb{R}$ with $v(x) \ne 0$ one has
\[
|u'(x)| \;=\; \frac{1}{k+1}\cdot\frac{|v'(x)|}{|v(x)|^{k/(k+1)}}
\;\le\; \frac{1}{k+1}\left\{ C(k)\sup_{x\in\mathbb{R}}\bigl|v^{(k+1)}(x)\bigr| \right\}^{1/(k+1)} .
\]
\end{remark}
