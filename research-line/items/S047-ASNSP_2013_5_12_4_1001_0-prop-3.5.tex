\section*{Proposition 3.5}

\textbf{Source.} M.~Ghisi, M.~Gobbino, \emph{Higher order Glaeser inequalities and optimal
regularity of roots of real functions}, Ann.\ Sc.\ Norm.\ Super.\ Pisa Cl.\ Sci.\ (5)
\textbf{XII} (2013), 1001--1021; printed page 1013.

\begin{proposition}
Let $k$ be a positive integer, let $\alpha\in(0,1]$, let $(a,b)\subseteq\mathbb{R}$ be an interval,
let $f:(a,b)\to\mathbb{R}$ be a continuous function, and let $g\in C^{k,\alpha}((a,b))$.

Let us assume that $f$ and $g$ satisfy (2.2), and that $g(x)\cdot g'(x)\neq 0$ for every
$x\in(a,b)$. Let us set for simplicity
\[
H := \Hold_\alpha\!\left(g^{(k)},(a,b)\right), \qquad \|g'\|_\infty := \|g'\|_{L^\infty((a,b))}.
\]
Then we have the following conclusions.

\emph{(1)} There exists a constant $C_1(k)$, depending only upon $k$, such that for every
$x\in(a,b)$ we have that
\begin{equation}
|f'(x)|^{k+\alpha} \;\le\; C_1(k)\max\left\{ H,\;
\|g'\|_\infty\cdot\left[\frac{b-a}{(x-a)(b-x)}\right]^{k+\alpha-1} \right\}. \tag{3.8}
\end{equation}

\emph{(2)} Let $p$ be defined by (2.5). Then there exists a constant $C_2(k)$, depending only upon
$k$, such that
\begin{equation}
\|f'\|_{p,w,(a,b)} \;\le\; C_2(k)\max\left\{ H^{1/(k+\alpha)}\cdot(b-a)^{1/p},\;
\|g'\|_\infty^{1/(k+\alpha)} \right\}. \tag{3.9}
\end{equation}
\end{proposition}

\subsection*{Proof (authors' text)}

Let us fix any point $x_0\in(a,b)$, and let $d := \min\{x_0-a,\,b-x_0\}$. We can apply
Proposition 3.4 to the function $v(x) := g(x)$ in the interval $(x_0-d,\,x_0+d)$. We obtain that
\[
|f'(x_0)|^{k+\alpha} \;\le\; \frac{|g'(x_0)|^{k+\alpha}}{|g(x_0)|^{k+\alpha-1}}
\;\le\; C(k)\max\left\{ H,\; \frac{|g'(x_0)|}{d^{\,k+\alpha-1}} \right\}.
\]
Since
\[
\frac{1}{d^{\,k+\alpha-1}} \;\le\; \left[\frac{b-a}{(x_0-a)(b-x_0)}\right]^{k+\alpha-1},
\]
conclusion (3.8) easily follows.

Let us consider now statement (2). From (3.8) and definition (2.5) of $p$, we have that
$|f'(x)| \le [C_1(k)]^{1/(k+\alpha)}\max\{\psi_1(x),\psi_2(x)\}$, with
\[
\psi_1(x) := H^{1/(k+\alpha)}, \qquad
\psi_2(x) := \|g'\|_\infty^{1/(k+\alpha)}\cdot\frac{(b-a)^{1/p}}{[(x-a)(b-x)]^{1/p}} .
\]
From statements (2) through (4) of Lemma 3.1 we easily obtain (3.9). \hfill$\square$
