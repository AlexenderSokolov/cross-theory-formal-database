\def\Xcfdiamond#1#2#3#4#5{%
  \node[Xcf vertex,
    label={[Xcf label]#4:{$#5$}}]
  (#1) at (#2,#3) {};
}

\tikzset{
  Xcf vertex/.style={
    circle,
    fill=black,
    inner sep=1.35pt,
    outer sep=0pt
  },
  Xcf label/.style={
    font=\scriptsize,
    inner sep=1pt,
    fill=white,
    text=black
  },
  Xcf picture/.style={
    x=.78cm,
    y=.78cm,
    line width=.45pt,
    label distance=2pt
  }
}

% Coordinates and edges common to all six panels.
\def\Xcfbase{%
  \path[use as bounding box]
    (-3.5,-2.65) rectangle (3.5,2.65);

  \foreach \n/\x/\y in {
    a0/-2.4/1.65,
    b0/-1.6/1.65,
    c0/-2/.65,
    a1/1.6/1.65,
    b1/2.4/1.65,
    c1/2/.65,
    a2/2.4/-1.65,
    b2/1.6/-1.65,
    c2/2/-.65,
    a3/-1.6/-1.65,
    b3/-2.4/-1.65,
    c3/-2/-.65
  }{
    \coordinate (\n) at (\x,\y);
  }

  \foreach \i in {0,...,3}{
    \draw (a\i)--(b\i)--(c\i)--cycle;
  }
}

% Draw each base vertex with only its color.
\def\Xcfcolors#1{%
  \foreach \n/\ang/\col in {#1}{
    \node[Xcf vertex,
      label={[Xcf label]\ang:{$\col$}}]
    at (\n) {};
  }
}
