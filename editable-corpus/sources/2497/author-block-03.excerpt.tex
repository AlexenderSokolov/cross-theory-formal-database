\begin{remark}\label{rem4-5}
It is proved in \cite{derksen2022g} that
\[
\frac{2 \sr_{\mathbb{K}}(T)}{d} \le \R^G_{\mathbb{K}}(T) = \R^G_{\overline{\mathbb{K}}}(T) \le \sr_{\overline{\mathbb{K}}}(T),
\]
where $\R^G_{\mathbb{K}}(T)$ is the G-stable rank of $T$.  This leads to the stability of the slice rank under field extensions with a better constant:
\begin{equation}\label{rem4-5:eq1}
\sr_{\mathbb{K}}(T)\le \frac{d}{2} \sr_{\overline{\mathbb{K}}}\left( T^{\overline{\mathbb{K}}} \right).
\end{equation}
\end{remark}
