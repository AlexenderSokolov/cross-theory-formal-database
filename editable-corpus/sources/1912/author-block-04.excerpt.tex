\section{Spherical vs. Iwahori--Whittaker}
\label{jep104local029}

\subsection{Equivariant perverse sheaves on \texorpdfstring{$\mathrm{Gr}$}{Gr}}
\label{jep104local032}

For $\lambda \in \boldsymbol{X}^\vee_+$, we will denote by
\[
\jepPubEuler{J}_!(\lambda,\Bbbk):= {}^{p}\jepPubEuler{H}^0 \bigl( (j_\lambda)_! \underline{\Bbbk}_{\mathrm{Gr}^\lambda}[\langle \lambda, 2\rho \rangle] \bigr), \quad \text{resp.} \quad \jepPubEuler{J}_*(\lambda,\Bbbk):= {}^{p}\jepPubEuler{H}^0 \bigl( (j_\lambda)_* \underline{\Bbbk}_{\mathrm{Gr}^\lambda}[\langle \lambda, 2\rho \rangle] \bigr),
\]
the standard, resp.~costandard, $G_{\jepPubScript{O}}$-equivariant perverse sheaf on $\mathrm{Gr}$ associated with $\lambda$.

We will also denote by $\jepPubEuler{J}_{!*}(\lambda,\Bbbk)$ the image of any generator of the free rank-$1$ $\Bbbk$-module
\[
\operatorname{Hom}_{\mathrm{Perv}_{G_{\jepPubScript{O}}}(\mathrm{Gr}, \Bbbk)}(\jepPubEuler{J}_!(\lambda,\Bbbk), \jepPubEuler{J}_*(\lambda,\Bbbk)).
\]
If $\Bbbk$ is a field then $\jepPubEuler{J}_{!*}(\lambda,\Bbbk)$ is a simple perverse sheaf, which is both the head of $\jepPubEuler{J}_!(\lambda,\Bbbk)$ and the socle of $\jepPubEuler{J}_*(\lambda,\Bbbk)$.

Recall the notion of highest weight category, whose definition is spelled out e.g.~in~\cite[Def.~7.1]{jep104ref40}. (These conditions are obvious extensions of those considered in~\cite[\S 3.2]{jep104ref14}, which were preceded by a related study in~\cite{jep104ref20}.)

\begin{lem}
\label{jep104local015}
Assume that $\Bbbk$ is a field.
The category $\mathrm{Perv}_{G_{\jepPubScript{O}}}(\mathrm{Gr}, \Bbbk)$ is a highest weight category with weight poset $(\boldsymbol{X}^\vee_+, \preccurlyeq)$, standard objects $\{\jepPubEuler{J}_!(\lambda,\Bbbk) : \lambda \in \boldsymbol{X}^\vee_+\}$, and costandard objects $\{\jepPubEuler{J}_*(\lambda,\Bbbk) : \lambda \in \boldsymbol{X}^\vee_+\}$. Moreover, if $\mathrm{char}(\Bbbk)=0$ then this category is semisimple.
\end{lem}

\begin{proof}
The first claim is an easy consequence of~\cite[Prop.~10.1(b)]{jep104ref36}; see~\cite[Prop.~1.12.4]{jep104ref11} for details.
If $\mathrm{char}(\Bbbk)=0$, the semisimplicity of the category $\mathrm{Perv}_{G_{\jepPubScript{O}}}(\mathrm{Gr}, \Bbbk)$ is well known: see~\cite[Prop.~1]{jep104ref27} (or~\cite[\S 1.4]{jep104ref11} for an expanded version).
\end{proof}

\begin{rmk}\phantomsection
\label{jep104local024}
\begin{enumerate}
\item
\label{jep104local014}
If $\Bbbk$ is a field of characteristic $0$, the semisimplicity of the category $\mathrm{Perv}_{G_{\jepPubScript{O}}}(\mathrm{Gr}, \Bbbk)$ implies in particular that the natural maps $\jepPubEuler{J}_!(\lambda,\Bbbk) \to \jepPubEuler{J}_{!*}(\lambda,\Bbbk) \to \jepPubEuler{J}_*(\lambda,\Bbbk)$ are isomorphisms.
\item
\label{jep104local011}
 For any coefficients $\Bbbk$, we have
 \[
  \operatorname{Hom}_{D^{\mathrm{b}} \mathrm{Perv}_{G_{\jepPubScript{O}}}(\mathrm{Gr},\Bbbk)} \bigl( \jepPubEuler{J}_!(\lambda,\Bbbk), \jepPubEuler{J}_*(\mu,\Bbbk)[n] \bigr)=
  \begin{cases}
   \Bbbk & \text{if $n=0$ and $\lambda=\mu$;} \\
    0 & \text{otherwise.}
\end{cases}
 \]
In fact, to prove this it suffices to prove the similar claim for perverse sheaves on $Z$, where $Z \subset \mathrm{Gr}$ is any closed finite union of $G_{\jepPubScript{O}}$-orbits containing $\mathrm{Gr}^\lambda$ and $\mathrm{Gr}^\mu$. In the case $\Bbbk$ is a field, this claim is a consequence of Lemma~\ref{jep104local015} (or rather its version for $Z$). The case when $\Bbbk$ is the ring of integers in a finite extension of $\jepPubBbb{Q}_\ell$ follows. Indeed, since the category $\mathrm{Perv}_{G_{\jepPubScript{O}}}(Z,\Bbbk)$ has enough projectives we can consider the complex of $\Bbbk$-modules
\[
 M=R\operatorname{Hom}_{\mathrm{Perv}_{G_{\jepPubScript{O}}}(Z,\Bbbk)}(\jepPubEuler{J}_!(\lambda,\Bbbk), \jepPubEuler{J}_*(\mu,\Bbbk)).
\]
If $\Bbbk_0$ is the residue field of $\Bbbk$, it is not difficult (using the results of~\cite[\S 8 \& \S 10]{jep104ref36}, and in particular the fact that $\Bbbk_0 \jepDerivedTensor_\Bbbk \jepPubEuler{J}_?(\lambda,\Bbbk) \cong \jepPubEuler{J}_?(\lambda,\Bbbk_0)$ for $? \in \{!,*\}$, see~\cite[Prop.~8.1]{jep104ref36}) to check that
\[
 \Bbbk_0 \jepDerivedTensor_\Bbbk M \cong R\operatorname{Hom}_{\mathrm{Perv}_{G_{\jepPubScript{O}}}(Z,\Bbbk_0)}(\jepPubEuler{J}_!(\lambda,\Bbbk_0), \jepPubEuler{J}_*(\mu,\Bbbk_0)).
\]
We deduce that the left-hand side is isomorphic to $\Bbbk_0$ in the derived category of $\Bbbk_0$-vector spaces; this implies that $M$ is isomorphic to $\Bbbk$ in the derived category of $\Bbbk$-modules.
\end{enumerate}
\end{rmk}

In Section~4 we will also encounter some $I^-_\mathrm{u}$-equivariant perverse sheaves on $\mathrm{Gr}$, where $I^-_\mathrm{u}$ is the pro-unipotent radical of $I^-$. In particular, we have
\[
\mathrm{Gr} = \mathop{\textstyle\bigsqcup}\limits_{\mu \in \boldsymbol{X}^\vee} I_\mathrm{u}^- \cdot L_\mu,
\]
and we will denote by $\Delta^\mathrm{Gr}_\mu(\Bbbk)$, resp.~$\nabla^\mathrm{Gr}_\mu(\Bbbk)$, the standard, resp.~costandard, perverse sheaf associated with $\mu$, i.e.~the $!$-direct image (resp.~$*$-direct image) of the constant perverse sheaf of rank $1$ on $I^-_\mathrm{u} \cdot L_\mu$. (These objects are perverse sheaves thanks to~\cite[Cor.~4.1.3]{jep104ref12}, because the orbits $I^-_\mathrm{u}\cdot L_\mu$ are affine spaces.)
