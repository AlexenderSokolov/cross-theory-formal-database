\begin{thm}\label{thm:b0ConcGen}
Let $f_1,f_2: \mb{N} \ra \mb{C}$ be multiplicative functions and let $h \geq 1$. Assume that
%
\begin{align}
\sum_{p: |f_1(p)| \neq 1} \frac{1}{p} = \infty. \label{eq:thinF1}
%
\end{align}
Let $a,b \in \mb{N}$, $c,d \in \mb{Z}$ with $ad-bc \neq 0$ and $(a,b) = (c,d) = 1$. Let also $C \in \mb{C} \bk \{0\}$. 
Then the set
$$
\{n \in \mb{N} : f_1(an+b) = Cf_2(cn+d) \neq 0\}
$$
has logarithmic density $0$, i.e., we have
\begin{equation}\label{eq:logDensZeroGen}
\frac{1}{\log x} \sum_{\ss{n \leq x \\ f_1(an+b)f_2(cn+d) \neq 0}} \frac{1_{f_1(an+b) = Cf_2(cn+d)}}{n} = o(1).
\end{equation}
%
%
%
%
%
%
%
%
\end{thm}
