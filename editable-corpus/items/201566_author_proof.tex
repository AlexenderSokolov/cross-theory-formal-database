	\begin{theorem}
		If
		\[
		\frac{1}{Y_7^*(q)} = q^{-7} Y_7(q)
		= \frac{(q^{3}, q^{65}; q^{68})_\infty}{(q^{31}, q^{37}; q^{68})_\infty}
		= \sum_{n=0}^{\infty} \xi_n q^n,
		\]
		then we have
		\[
		\xi_{34n+28} = 0.
		\]
	\end{theorem}
