\noindent\textit{Allowed standard prerequisites (editorial).}\begin{itemize}
\item Bounded elementary substructure. A\$\textbackslash{}leq\$n B means that A is elementary in B already after adding any set of at most n elements of B.. It is the relation whose transitivity is asserted.
\item Diamond condition. For unconfined A1, A2 and B containing A1 with A2 elementary in B, the intersection of A1 and A2 is elementary in A1.. It is the tool that splits the added set.
\item Closure of unconfined systems. The unconfined systems are closed under the relevant operations, so the intermediate systems of the argument remain unconfined.. It lets the diamond condition be applied.
\end{itemize}\medskip
\begin{lemma}\label{diamond} Let $A,A_1,A_2 \in \Cun$. Then
	\begin{enumerate}
		\item $\emptyset \leq A$  
	\item If $B\in \Cun$ is such that $A_1\subseteq B$ and $A_2\leq B$, then $A_1\cap A_2\leq A_1$.
	\end{enumerate}
\end{lemma}

\begin{lemma}\label{sat3}  If  $A\leq^n B$ and $B\leq^n C$, then $A\leq^n C$.
\end{lemma}

\begin{proof}  Assume $X\subseteq C \smallsetminus A$  and $|X|\leq n$. We check that  $A\leq A\cup X$. Since $B\leq^n C$, we have $B\leq B\cup X$. By Lemma~\ref{diamond},   $A\cup (X\cap B)\leq A\cup X$.
	Since $A\leq^n B$, we know that $A\leq A\cup (X\cap B)$.  Hence  $A\leq A\cup X$.
\end{proof}
