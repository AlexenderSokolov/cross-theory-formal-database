\begin{lemma}
\label{lemma-finite-goes-up}
Let $K/E/F$ be a tower of algebraic field extensions.
If $K$ is finite over $F$, then $K$ is finite over $E$.
\end{lemma}

\begin{proof}
Direct from the definition.
\end{proof}
