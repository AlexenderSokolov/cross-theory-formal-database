\noindent\textit{Allowed standard prerequisites (editorial).}\begin{itemize}
\item The function rho. rho(n) is the least divisor of n that is at least the square root of n.. It is the function whose average is estimated.
\item The Wasserstein-type identity for Lipschitz statistics. For a bounded Lipschitz function of the normalised logarithmic parts, the expectation of the truncated version equals the expectation of the same function of the Dirichlet limit plus an error of order 1/log x.. It is applied twice, to a specific Lipschitz function and to the divisors themselves.
\item Partial summation. Sums of the form sum over n up to x of a quantity divided by log n can be converted into Abel sums.. It passes from the normalised sum to the unnormalised asymptotic.
\end{itemize}\medskip
\begin{equation}
\label{eq:dual wasserstein}
\EE\bgg[\one_{N_x \ge 2}\cdot \psi\bg(\tfrac{\log P_1}{\log N_x}, \tfrac{\log P_2}{\log N_x}, \ldots\bg)\bgg] = \EE\bg[\psi\bg(V_1, V_2, \ldots\bg)\bg] + O_\psi(1/\log x).
\end{equation}

\begin{cor}
Let $\rho(n) \coloneqq \min\{d|n : d\ge \sqrt n\}$. Then there is a constant $c \in (1/2,1)$ such that
\[
\sum_{n\le x} \log \rho(n) = cx\log x + O(x) \qquad (x \ge 1).
\]
\end{cor}

\begin{proof}
Let
\[
\psi(v_1, v_2, \ldots) \coloneqq \inf \big( \mathcal D(v_1, v_2, \ldots) \cap [\tfrac{1}{2}, 1]\big)
\]
with $\mathcal D(v_1, v_2, \ldots) \coloneqq \{\sum_{i \in B} v_i : B\subseteq \mathbb N\}$. It~is easy to check that this is a $1$-Lipschitz function. Note that the left-hand side of \eqref{eq:dual wasserstein} with this definition for $\psi$ is exactly equal to
\[
\frac{1}{\floor x} \sum_{2 \le n \le x} \frac{\log \rho(n)}{\log n}.
\]
With \eqref{eq:dual wasserstein} and partial summation, we~prove the corollary with $c = \EE[\psi(\bV)]$.
\end{proof}
