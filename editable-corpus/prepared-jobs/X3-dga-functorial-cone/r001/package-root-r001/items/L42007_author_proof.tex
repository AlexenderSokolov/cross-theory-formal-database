\noindent\textit{Allowed standard prerequisites (editorial).}\begin{itemize}
\item Cone of a morphism of dg modules. The cone C(f) of f : X -> Y is the dg module Y direct sum X[1] carrying the inclusion i, the projection p and the differential built from d\_Y, f and the shifted d\_X.. The maps i\_1, p\_1, i\_2, p\_2 and the differentials of the two cones are what the constructed c must be compatible with (source lines 655-658, 672-680).
\item Homotopy between morphisms of dg modules. A homotopy between two morphisms of dg modules is a graded map h of degree minus one such that their difference equals d h plus h d.. The homotopy h between f\_2 a and b f\_1 becomes the off-diagonal entry of the matrix defining c and is what makes c a morphism of dg modules (source lines 666-672).
\item Morphism of triangles in the homotopy category. A morphism between two distinguished triangles is a triple of morphisms making the two relevant squares commute in the homotopy category.. The conclusion is that the triple (a, b, c) is such a morphism (source lines 674-680).
\end{itemize}\medskip
\begin{lemma}
\label{lemma-functorial-cone}
Let $(A, \text{d})$ be a differential graded algebra.
Suppose that
$$
\xymatrix{
K_1 \ar[r]_{f_1} \ar[d]_a & L_1 \ar[d]^b \\
K_2 \ar[r]^{f_2} & L_2
}
$$
is a diagram of homomorphisms of differential graded $A$-modules which is
commutative up to homotopy.
Then there exists a morphism $c : C(f_1) \to C(f_2)$ which gives rise to
a morphism of triangles
$$
(a, b, c) : (K_1, L_1, C(f_1), f_1, i_1, p_1) \to
(K_1, L_1, C(f_1), f_2, i_2, p_2)
$$
in $K(\text{Mod}_{(A, \text{d})})$.
\end{lemma}

\begin{proof}
Let $h : K_1 \to L_2$ be a homotopy between $f_2 \circ a$ and $b \circ f_1$.
Define $c$ by the matrix
$$
c =
\left(
\begin{matrix}
b & h \\
0 & a
\end{matrix}
\right) :
L_1 \oplus K_1 \to L_2 \oplus K_2
$$
A matrix computation show that $c$ is a morphism of differential
graded modules. It is trivial that $c \circ i_1 = i_2 \circ b$, and it is
trivial also to check that $p_2 \circ c = a \circ p_1$.
\end{proof}
