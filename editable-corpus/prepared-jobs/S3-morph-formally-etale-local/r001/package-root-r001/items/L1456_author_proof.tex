\begin{lemma}
\label{lemma-formally-etale}
Let $f : X \to Y$ be a morphism of schemes. The following are equivalent:
\begin{enumerate}
\item $f$ is formally \'etale,
\item for every $x \in X$ there exist opens $x \in U \subset X$ and
$f(x) \in V \subset Y$ with $f(U) \subset V$ such that
$f|_U : U \to V$ is formally \'etale,
\item for every pair of affine opens $U \subset X$ and $V \subset Y$
with $f(U) \subset V$ the ring map $\mathcal{O}_Y(V) \to \mathcal{O}_X(U)$
is formally \'etale, and
\item there exists an affine open covering $Y = \bigcup V_j$ and
for each $j$ an affine open covering $f^{-1}(V_j) = \bigcup U_{ji}$
such that $\mathcal{O}_Y(V) \to \mathcal{O}_X(U)$ is a formally \'etale
ring map for all $j$ and $i$.
\end{enumerate}
\end{lemma}

\begin{proof}
Combining Lemmas \ref{lemma-formally-etale-on-opens} and
\ref{lemma-affine-formally-etale} we see that (1) $\Rightarrow$ (3).
It is immediate that (3) $\Rightarrow$ (4).
We have (4) $\Rightarrow$ (2) by Lemma \ref{lemma-affine-formally-etale}.
If (2) holds, then $\Omega_{X/S}$ and $\mathcal{C}_{X/S}$ vanish
in a neighbourhood of every point by
Lemma \ref{lemma-characterize-formally-etale}
(this also uses Morphisms, Lemma
\ref{morphisms-lemma-differentials-restrict-open} and
Lemma \ref{lemma-universal-thickening-localize}) whereupon
Lemma \ref{lemma-characterize-formally-etale} tells us that $f$
is formally unramified.
\end{proof}
