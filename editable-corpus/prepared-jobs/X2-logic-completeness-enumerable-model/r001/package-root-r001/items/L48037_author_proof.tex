\noindent\textit{Editorial scope.} Statement and complete author proof are included verbatim as the span lines around ollabel thm:downward-ls of the pinned original source; no text is rewritten or paraphrased.\par\medskip
\noindent\textit{Allowed standard prerequisites (editorial).}\begin{itemize}
\item the completeness theorem's term-model construction
\item the term set of a countable language and its cardinality
\item finite and denumerable domains
\end{itemize}\medskip
\begin{thm} 
\ollabel{thm:downward-ls} If $\Gamma$ is consistent then it has
!!a{enumerable} model, i.e., it is satisfiable in !!a{structure}
whose domain is either finite or !!{denumerable}.
\end{thm}

\begin{proof}
If $\Gamma$ is consistent, the !!{structure}~$\Struct M$ delivered by
the proof of the completeness theorem has a domain $\Domain{M}$ that
is no larger than the set of the terms of the language~$\Lang L$. So
$\Struct M$ is at most !!{denumerable}.
\end{proof}
